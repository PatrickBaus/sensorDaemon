36# A host with an optional port: Either "[IPv6 address]:port" or "hostname:port". The port is optional. 

37HOST_PORT_REGEX = re.compile(r"^(?:\[(?P<ipv6>[^\]]+)\]|(?P<hostname>[^:\[\]]+))(?::(?P<port>[0-9]{1,5}))?$") 

38# A hostname label as defined by RFC 1035 (https://www.rfc-editor.org/rfc/rfc1035), but also allowing underscores, 

39# because they are commonly used in container names, e.g. by Docker Compose. 

40HOSTNAME_LABEL_REGEX = re.compile(r"^[0-9A-Za-z_](?:[0-9A-Za-z_-]{0,61}[0-9A-Za-z_])?$") 

41DEFAULT_MQTT_PORT = 1883

44def parse_host(host: str) -> tuple[str, int]: 

45 """ 

46 Parse a hostname, IPv4 or IPv6 address with an optional port. IPv6 addresses must be enclosed in brackets. 

47 

48 Parameters 

49 ---------- 

50 host: str 

51 The host to parse, e.g. "example.com:1883", "mqtt_broker", "192.168.1.2:1883" or "[::1]:1883" 

52 

53 Returns 

54 ------- 

55 tuple of str and int 

56 The hostname or address and the port. If no port or port 0 is given, the MQTT default port 1883 is returned. 

57 

58 Raises 

59 ------ 

60 ValueError 

61 If the host is not a valid hostname, address or port. 

62 """ 

63 match = HOST_PORT_REGEX.match(host) 

64 if match is None: 

65 raise ValueError(f"'{host}' is not a valid hostname.") 

66 port = int(match.group("port")) if match.group("port") else 0 

67 if port > 65535: 

68 raise ValueError(f"'{host}' has an invalid port number.") 

69 port = port or DEFAULT_MQTT_PORT 

70 

71 if match.group("ipv6") is not None: 

72 try: 

73 return str(ipaddress.IPv6Address(match.group("ipv6"))), port 

74 except ValueError: 

75 raise ValueError(f"'{host}' is not a valid IPv6 address.") from None 

76 

77 hostname = match.group("hostname") 

78 labels = hostname[:-1].split(".") if hostname.endswith(".") else hostname.split(".") 

79 if len(hostname) > 255 or not all(HOSTNAME_LABEL_REGEX.match(label) for label in labels): 

80 raise ValueError(f"'{host}' is not a valid hostname.") 

81 return hostname, port

368 """ 

369 The main task, that spawns all workers. 

370 """
